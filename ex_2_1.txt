Exercise 2.1  Using Apprendix A, translate each of the following pseudocode
expressions into MIPS assembly langauge:

(a) t3 = t4 + t5 - t6;
(b) s3 = t2 / (s1 - 54321);
(c) sp = sp - 16;
(d) cout << t3;
(e) cin >> t0;
(f) a0 = &array;
(g) t8 = Mem(a0);
(h) Mem(a0+16) = 32768;
(i) cout << "Hello World";
(j) If (t0 < 0) then t7 = 0 - t0 else t7 = t0;
(k) while (t0 != 0) {s1 = s1 + t0; t2 = t2 + 4; t0 = Mem(t2)};
(l) for (t = 99; t1 > 0; t1 = t1 - 1) v0 = v0 + 1;
(m) t0 = 2147483647 - 2147483648;
(n) s0 = -1 * s0;
(o) s1 = s1 * a0;
(p) s2 = srt(s0^2 + 56) / a3;
(q) s3 = s1 - s2 / s3;
(r) s4 = s4 * 8;
(s) s5 = 7 * s5;

Solution 2.1

(a)
sub $t5, $t5, $t6
add $t3, $t4, $t5

(b)
li $t0, 54321
sub $s1, $s1, $t0
div $s3, $t2, $s1

(c)
sub $sp, $sp, 16

(d)
li $v0, 1
move $a0, $t3
syscall

(e)
li $v0, 5
syscall
move $t0, $v0

(f)
        .data
array:  .space 4096
        .text
la $a0, array

(g) t8 = Mem(a0);
lw $t8, Mem($a0)


(h) Mem(a0+16) = 32768;
li $t1, 32768
sw $t1, 16($a0)

(i) cout << "Hello World";
        .data
Msg:    .asciiz     "Hello World"
        .text
li      $v0, 4
la      $a0, Msg
syscall

(j) If (t0 < 0) then t7 = 0 - t0 else t7 = t0;
bgez $t0, else
sub $t0, $zero, $t0
move $t7, $t0
b next
else:
move $t7, $zero
next:

(k) while (t0 != 0) {s1 = s1 + t0; t2 = t2 + 4; t0 = Mem(t2)};
while:
    beq     $t0, $zero, done
    add     $s1, $s1, $t0
    addi    $t2, $t2, 4
    move    $t0, Mem($t2)
done:

(l) for (t1 = 99; t1 > 0; t1 = t1 - 1) v0 = v0 + 1;
    li      $t1, 99
loop:
    sub     $t1, $t1, 1
    add     $v0, $v0, 1
    bgtz    $t1, loop

(m) t0 = 2147483647 - 2147483648;
li      $t0, 2147483647
li      $t1, 2147483648
subu    $t0, $t0, $t1

(n) s0 = -1 * s0;
sub     $s0, $zero, $s0

(o) s1 = s1 * a0;
mul     $s1, $s1, $a0

(p) s2 = sqrt(s0^2 + 56) / a3;
mul     $s0, $s0
add     $s0, $s0, 56
jal     srt
move    $s0, $v0
div     $s0, $a3
move    $s2, $s0

(q) s3 = s1 - s2 / s3;
div     $s2, $s3
sub     $s1, $s1, $s2
move    $s3, $s1

(r) s4 = s4 * 8;
sll     $s4, $s4, 3

(s) s5 = 7 * s5;
li      $s4, 7
mul     $s5, $s5, $s4
