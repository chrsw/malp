Exercise 2.1  Using Apprendix A, translate each of the following pseudocode
expressions into MIPS assembly langauge:

(a) t3 = t4 + t5 - t6;
(b) s3 = t2 / (s1 - 54321);
(c) sp = sp - 16;
(d) cout << t3;
(e) cin >> t0;
(f) a0 = &array;
(g) t8 = Mem(a0);
(h) Mem(a0+16) = 32768;
(i) cout << "Hello World";
(j) If (t0 < 0) then t7 = 0 - t0 else t7 = t0;
(k) while (t0 != 0) {s1 = s1 + t0; t2 = t2 + 4; t0 = Mem(t2)};

Solution 2.1

(a)
sub $t5, $t5, $t6
add $t3, $t4, $t5

(b)
li $t0, 54321
sub $s1, $s1, $t0
div $s3, $t2, $s1

(c)
sub $sp, $sp, 16

(d)
li $v0, 1
move $a0, $t3
syscall

(e)
li $v0, 5
syscall
move $t0, $v0

(f)
        .data
array:  .space 4096
        .text
la $a0, array

(g) t8 = Mem(a0);
lw $t8, Mem($a0)


(h) Mem(a0+16) = 32768;
li $t1, 32768
sw $t1, 16($a0)

(i) cout << "Hello World";
        .data
Msg:    .asciiz     "Hello World"
        .text
li      $v0, 4
la      $a0, Msg
syscall

(j) If (t0 < 0) then t7 = 0 - t0 else t7 = t0;
bgez $t0, else
sub $t0, $zero, $t0
move $t7, $t0
b next
else:
move $t7, $zero
next:

(k) while (t0 != 0) {s1 = s1 + t0; t2 = t2 + 4; t0 = Mem(t2)};
while:
    beq     $t0, $zero, done
    add     $s1, $s1, $t0
    addi    $t2, $t2, 4
    move    $t0, Mem($t2)
done:
